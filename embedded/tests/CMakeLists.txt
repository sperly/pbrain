cmake_minimum_required(VERSION 3.16)
project(pocketbrain_host_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
add_executable(core_tests test_core.c ../components/pb_core/pb_core.c ../components/pb_core/pb_journal.c)
target_include_directories(core_tests PRIVATE ../components/pb_core/include)
target_compile_options(core_tests PRIVATE -Wall -Wextra -Werror -Wconversion -UNDEBUG)
add_test(NAME core_tests COMMAND core_tests)
add_executable(link_tests test_link.c ../components/pb_core/pb_link.c)
target_include_directories(link_tests PRIVATE ../components/pb_core/include)
target_compile_options(link_tests PRIVATE -Wall -Wextra -Werror -Wconversion -UNDEBUG)
add_test(NAME link_tests COMMAND link_tests)
